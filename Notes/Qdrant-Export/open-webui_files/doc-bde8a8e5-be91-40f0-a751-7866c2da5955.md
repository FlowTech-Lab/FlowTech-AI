# lib/view.rb — I/O terminal
class View
  def form_entity
    print "Auteur: "; author = STDIN.gets&.chomp
    print "Contenu: "; content = STDIN.gets&.chomp
    { author: author, content: content }
  end
  def index(items)
    items.each_with_index { |e, i| puts "#{i+1}. {author: #{e.author}} — {content: #{e.content}}" }
  end
  def ask_index; print "Index à supprimer: "; STDIN.gets.to_i - 1; end
  def notice(msg); puts "[OK] {#{msg}}"; end
end

require "csv"
class Entity
  attr_reader :author, :content
  DB_PATH = "db/gossip.csv"
  def initialize(author, content); @author = author; @content = content; end
  def save; CSV.open(DB_PATH, "a") { |csv| csv << [author, content] }; end
  def self.all
    return [] unless File.exist?(DB_PATH)
