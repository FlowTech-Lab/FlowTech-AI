### ✍️ Snippets essentiels
```bash
# Objectif: créer une clé ed25519 sans passphrase
ssh-keygen -t ed25519 -C "EMAIL" -f ~/.ssh/id_ed25519 -N ""

# Objectif: activer l'agent et charger la clé
eval "$(ssh-agent -s)"; ssh-add ~/.ssh/id_ed25519

# Objectif: fichier ~/.ssh/config pour GitHub
# Simplifie l'usage de l'agent et des clés
printf "%s\n" "Host github.com
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519
  IdentitiesOnly yes" >> ~/.ssh/config
chmod 600 ~/.ssh/config

# Objectif: vérifier l'empreinte du host et l'accès
ssh -Tv git@github.com

# Objectif: basculer un repo existant en SSH
git remote -v
git remote set-url origin git@github.com:ORG_OR_USER/REPO.git
git push -u origin main
```
