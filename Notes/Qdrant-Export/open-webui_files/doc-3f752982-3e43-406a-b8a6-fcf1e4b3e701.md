    return puts "Tous les joueurs sont déjà en vue" if @enemies_in_sight.size >= @players_left
    roll = rand(1..6)
    spawn = (roll == 1 ? 0 : roll <= 4 ? 1 : 2)

      break if @enemies_in_sight.size >= @players_left
      id = rand(1000..9999)
