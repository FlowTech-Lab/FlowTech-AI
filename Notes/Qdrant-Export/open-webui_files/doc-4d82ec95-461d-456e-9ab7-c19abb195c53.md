# lib/app/saver.rb — sauvegardes multi-formats
require "json"; require "csv"
class Saver
  def save_as_json(data, path: "db/emails.json")
    File.write(path, JSON.pretty_generate(data))
  end
  def save_as_csv(data, path: "db/emails.csv")
    CSV.open(path, "w") do |csv|
      csv << %w[city email]
      data.each do |k, v|
        csv << [k, v]
      end
    end
  end
end
