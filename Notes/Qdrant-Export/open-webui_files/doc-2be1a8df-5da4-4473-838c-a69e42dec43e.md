def self.all
    return [] unless File.exist?(DB_PATH)
    CSV.read(DB_PATH).map { |a, c| new(a, c) }
  end
  def self.delete_at(idx)
    rows = all
    rows.delete_at(idx)
    File.write(DB_PATH, rows.map { |e| [e.author, e.content] }.map { |r| r.join(",") }.join("\n") + (rows.empty? ? "" : "\n"))
  end
end
