# Objectif: compter les occurrences des mots d'un dictionnaire dans un corpus
def word_counter(corpus, dictionnary)
  return {} unless corpus.is_a?(String) && dictionnary.is_a?(Array)
  tokens = corpus.downcase
  dictionnary.each_with_object({}) do |w, h|
    next unless w.is_a?(String)
    count = tokens.scan(Regexp.new(Regexp.escape(w.downcase))).size
    h[w] = count if count.positive?
  end
end
