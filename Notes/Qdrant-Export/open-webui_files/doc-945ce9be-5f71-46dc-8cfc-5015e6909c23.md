  def search_weapon
    lvl = rand(1..6)
    puts "Tu as trouvé une arme de niveau #{lvl}"
    if lvl > @weapon_level then @weapon_level = lvl; puts "Youhou ! meilleure arme, tu la prends."
    else puts "Pas mieux que l'actuelle."
    end
  end
  def search_health_pack
    roll = rand(1..6)
    case roll
    when 1 then puts "Tu n'as rien trouvé..."
    when 2..5 then @life_points = [@life_points + 50, 100].max && puts "Bravo, +50 PV !"
    when 6 then @life_points = [@life_points + 80, 100].max && puts "Waow, +80 PV !"
    end
  end
end
