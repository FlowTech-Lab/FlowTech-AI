# app_2.rb — menu interactif basique
require 'bundler'; Bundler.require
require_relative 'lib/player'
user = HumanPlayer.new("INPUT")
enemies = [Player.new("Bot1"), Player.new("Bot2")]
while user.life_points > 0 && enemies.any? { |e| e.life_points > 0 }
  user.show_state
  puts "a) arme  s) soin"; enemies.each_with_index { |e,i| puts "#{i}) #{e.name} (#{e.life_points} PV)" }
  case STDIN.gets&.chomp
  when 'a' then user.search_weapon
  when 's' then user.search_health_pack
  when '0','1'
    idx = _ = $_.to_i rescue 0
    target = enemies[idx]; user.attacks(target) if target&.life_points&.positive?
  end
  puts "Les ennemis ripostent !"
  enemies.each { |e| e.attacks(user) if e.life_points > 0 }
end
puts "La partie est finie"; puts(user.life_points > 0 ? "BRAVO !" : "Perdu.")
