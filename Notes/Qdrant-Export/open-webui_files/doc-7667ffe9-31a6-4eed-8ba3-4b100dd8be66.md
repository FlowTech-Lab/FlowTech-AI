# Objectif: boucle de conversation simple en CLI avec historique minimal
history = []
loop do
  print "Vous: "; input = STDIN.gets&.strip
  break if input.nil? || input.downcase == "stop"
  history << "Utilisateur: #{input}"
  prompt = (history + ["Assistant:"]).join("\n")
  resp = HTTP.headers(
    "Content-Type" => "application/json",
    "Authorization" => "Bearer #{ENV["OPENAI_API_KEY"]}"
  ).post("https://api.openai.com/v1/completions", json: {
    prompt: prompt, max_tokens: 200, temperature: 0.2, model: "babbage-002", n: 1
  })
  out = JSON.parse(resp.body.to_s).dig("choices", 0, "text").to_s.strip
  puts "IA: #{out}"
  history << "Assistant: #{out}"
end
