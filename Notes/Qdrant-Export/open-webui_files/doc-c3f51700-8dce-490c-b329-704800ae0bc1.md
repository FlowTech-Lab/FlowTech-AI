n = gets.to_i.clamp(1, 25)
1.upto(n) { |i| puts "#" * i }

# exo_16: pyramide montante (droite alignée)
n = gets.to_i.clamp(1, 25)
1.upto(n) { |i| puts " " * (n - i) + "#" * i }

# exo_17: pyramide full (montante puis descendante, centrée)
n = gets.to_i.clamp(1, 25)
1.upto(n) { |i| puts ("#" * (2*i - 1)).center(2*n - 1) }
