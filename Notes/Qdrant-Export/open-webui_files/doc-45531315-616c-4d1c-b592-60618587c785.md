SELECT a.title, u.name

-- N-N via table de jointure
SELECT a.title, c.title AS category
FROM article_categories ac
JOIN articles a ON a.id = ac.article_id
JOIN categories c ON c.id = ac.category_id;

-- Agrégat + totalisation d'un album
SELECT al.title, SUM(t.milliseconds)/1000 AS total_seconds, al.price
FROM albums al
JOIN tracks t ON t.album_id = al.id
WHERE al.title = 'INPUT'
GROUP BY al.id;
