# game.rb — squelette v3
class Game
  attr_accessor :human_player, :enemies_in_sight, :players_left
  def initialize(name)
    @human_player = HumanPlayer.new(name)
    @enemies_in_sight = []
    @players_left = 10
  end
  def kill_player(player) = @enemies_in_sight.delete(player)
  def is_still_ongoing? = @human_player.life_points > 0 && (@players_left > 0 || @enemies_in_sight.any?)
  def show_players
    @human_player.show_state
    puts "Bots restants: #{@players_left}, en vue: #{@enemies_in_sight.size}"
  end
  def menu
    puts "a) arme  s) soin"
    @enemies_in_sight.each_with_index { |e,i| puts "#{i}) #{e.name} (#{e.life_points} PV)" }
  end
  def menu_choice(input)
    case input
    when 'a' then @human_player.search_weapon
    when 's' then @human_player.search_health_pack
    else
      idx = input.to_i
      target = @enemies_in_sight[idx]
      if target
        @human_player.attacks(target)
        kill_player(target) if target.life_points <= 0
      end
    end
  end
