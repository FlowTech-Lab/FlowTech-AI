class Game
  attr_reader :board, :players, :status, :current
  def initialize(p1_name, p2_name)
    @board = Board.new
    @players = [Player.new(p1_name, "X"), Player.new(p2_name, "O")]
    @current = players.first; @status = :in_progress
  end
  def turn(input_pos)
    return status unless status == :in_progress
    return :error unless board.valid_move?(input_pos)
    board.play!(input_pos, current.symbol)
    if (w = board.winner) then @status = :win; return w
    elsif board.full? then @status = :draw; return :draw
