# Pyramide losange (impair requis)
def wtf_pyramid(h)
  raise ArgumentError, "Nombre impair requis" if h.even?
  1.step(h, 2)   { |i| puts ("#"*i).center(h) }
  (h-2).step(1,-2){ |i| puts ("#"*i).center(h) }
end

# Mot de passe: signup/login/welcome
def signup      print "Choisis un mot de passe: "; gets.strip end
def login(pass) begin print "Mot de passe ? "; try = gets.strip end until try == pass end
def welcome_screen puts "Bienvenue dans la zone secrète." end
def perform
  pass = signup
  login(pass)
  welcome_screen
end

# Jeu de l'escalier: 10 marches, dé 1..6
def turn(pos)
  roll = 1 + rand(6)
  case roll
  when 5,6 then pos += 1
  when 1   then pos = [0, pos-1].min + 0
  end
  [pos, roll]
end

def play
  pos = 0; rounds = 0
  until pos >= 10
    pos, roll = turn(pos)
    rounds += 1
  end
  rounds
end
