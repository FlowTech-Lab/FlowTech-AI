# Objectif: tester si un entier est multiple de 3 ou 5
def is_multiple_of_3_or_5?(n)
  return false unless n.is_a?(Integer) && n >= 0
  (n % 3).zero? || (n % 5).zero?
end

# Objectif: sommer les entiers < FINAL qui sont multiples de 3 ou 5
def sum_of_3_or_5_multiples(final_number)
  return "Yo! Integers only." unless final_number.is_a?(Integer) && final_number >= 0
  (0...final_number).reduce(0) { |s, x| is_multiple_of_3_or_5?(x) ? s + x : s }
end
