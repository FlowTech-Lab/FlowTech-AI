# Objectif: générer un labyrinthe correct (correction générique du squelette)
class Maze
  def initialize(width, height)
    @w, @h = width, height
    @grid = Array.new(@h * 2 + 1) { Array.new(@w * 2 + 1, "#") }
    carve(1, 1)
    @grid[1][0] = " "
    @grid[@h * 2 - 1][@w * 2] = " "
  end
  def display; @grid.each { |row| puts row.join }; end
  private
  def carve(x, y)
    @grid[y][x] = " "
    [[0,-2],[0,2],[-2,0],[2,0]].shuffle.each do |dx, dy|
      nx, ny = x + dx, y + dy
      next unless ny.between?(1, @h*2-1) && nx.between?(1, @w*2-1)
      next unless @grid[ny][nx] == "#"
      @grid[y + dy/2][x + dx/2] = " "
      carve(nx, ny)
    end
  end
end
# Usage
# Maze.new(10,10).display
