```rb
# Model avec validations et scopes
class Movie < ApplicationRecord
  validates :name, :year, :genre, :director, presence: true
  validates :allocine_rating, numericality: { allow_nil: true, greater_than_or_equal_to: 0, less_than_or_equal_to: 5 }
  scope :seen,  -> { where(already_seen: true) }
  scope :year_between, ->(a,b) { where(year: a..b) }
end
```
