# Objectif: affichage terminal simple
class Show
  def self.render(board)
    rows = [%w[A1 A2 A3], %w[B1 B2 B3], %w[C1 C2 C3]].map do |r|
      r.map { |p| board[p].value == " " ? p : board[p].value }.join(" | ")
    end
    system("clear") || system("cls")
    puts "   1   2   3"; puts "A  #{rows[0]}"; puts "B  #{rows[1]}"; puts "C  #{rows[2]}"
  end
end
