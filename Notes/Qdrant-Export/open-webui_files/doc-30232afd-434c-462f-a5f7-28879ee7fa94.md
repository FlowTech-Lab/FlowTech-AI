# Objectif: Model simple CSV
require 'csv'
class Gossip
  attr_reader :author, :content
  def initialize(author, content)
    @author  = author
    @content = content
  end
  def save
    CSV.open('db/gossip.csv', 'ab') { |csv| csv << [author, content, Time.now.utc.iso8601] }
  end
  def self.all
    return [] unless File.exist?('db/gossip.csv')
    CSV.read('db/gossip.csv').map { |a,c,_t| new(a, c) }
  end
  def self.find(id)
    all[id.to_i]
  end
  def self.update(id, author:, content:)
    rows = CSV.read('db/gossip.csv')
    rows[id.to_i] = [author, content, Time.now.utc.iso8601]
    CSV.open('db/gossip.csv', 'wb') { |csv| rows.each { |r| csv << r } }
  end
end
