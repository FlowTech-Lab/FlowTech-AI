# Player de base, PV 10, attaque aléatoire 1..6
class Player
  attr_accessor :name, :life_points
  def initialize(name) = (@name, @life_points = name, 10)
  def show_state = puts "#{@name} a #{@life_points} points de vie"
  def gets_damage(dmg)
    @life_points -= dmg
    puts "le joueur #{@name} a été tué !" if @life_points <= 0
  end
  def compute_damage = rand(1..6)
  def attacks(other)
    puts "#{@name} attaque #{other.name}"
    dmg = compute_damage
    other.gets_damage(dmg)
    puts "il lui inflige #{dmg} points de dommages"
  end
end
