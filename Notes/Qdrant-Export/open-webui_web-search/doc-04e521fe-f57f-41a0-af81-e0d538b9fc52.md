System-wide configuration requires root privileges:# Create system-wide configuration
sudo tee /etc/environment << EOF
OLLAMA_NUM_PARALLEL=4
OLLAMA_HOST=0.0.0.0
OLLAMA_MODELS=/opt/ollama-models
EOF
Windows ConfigurationConfigure environment variables through System Properties:# Command prompt method
setx OLLAMA_NUM_PARALLEL 4
setx OLLAMA_HOST 0.0.0.0
setx OLLAMA_MODELS "C:\OllamaModels"

# PowerShell method
[Environment]::SetEnvironmentVariable("OLLAMA_NUM_PARALLEL", "4", "User")
[Environment]::SetEnvironmentVariable("OLLAMA_HOST", "0.0.0.0", "User")
