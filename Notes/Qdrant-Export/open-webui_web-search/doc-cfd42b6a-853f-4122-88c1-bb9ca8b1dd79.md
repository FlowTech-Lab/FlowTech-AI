Node.js IntegrationSet up Ollama for Node.js applications:// Set environment variables
process.env.OLLAMA_HOST = '0.0.0.0:11434';
process.env.OLLAMA_NUM_PARALLEL = '6';

const ollama = require('ollama');

// Configure client
const client = new ollama.Ollama({
    host: 'http://localhost:11434'
});

// Make requests with optimized configuration
async function generateResponse(prompt) {
    const response = await client.generate({
        model: 'llama2',
        prompt: prompt,
        stream: false
    });
    return response.response;
}
